    76→    76→export function AdminQuestions() {
    77→    77→  const [classes, setClasses] = useState<AcademicClass[]>([]);
    78→    78→  const [subjects, setSubjects] = useState<AcademicSubject[]>([]);
    79→    79→  const [topics, setTopics] = useState<AcademicTopic[]>([]);
    80→    80→
    81→    81→  const [filterClassId, setFilterClassId] = useState<string>("all");
    82→    82→  const [filterSubjectId, setFilterSubjectId] = useState<string>("all");
    83→    83→  const [filterTopicId, setFilterTopicId] = useState<string>("all");
    84→    84→  const [filterStatus, setFilterStatus] = useState<string>("all");
    85→    85→  const [search, setSearch] = useState("");
    86→    86→
    87→    87→  const [questions, setQuestions] = useState<QuestionWithChain[] | null>(null);
    88→    88→  const [loading, setLoading] = useState(true);
    89→    89→  const [editing, setEditing] = useState<QuestionWithChain | null>(null);
    90→    90→  const [creating, setCreating] = useState(false);
    91→    91→
    92→    92→  // Load classes once for the class filter dropdown
    93→    93→  useEffect(() => {

   100→   100→      } catch {
   101→   101→        // ignore
   102→   102→      }
   103→   103→    }
   104→   104→    loadClasses();
   105→   105→  }, []);
   106→   106→
   107→   107→  // Load subjects when filter class changes (dependent dropdown)
   108→   108→  useEffect(() => {
   109→   109→    if (filterClassId === "all") {
   110→   110→      setSubjects([]);
   111→   111→      setFilterSubjectId("all");
   112→   112→      return;
   113→   113→    }
   114→   114→    let cancelled = false;
   115→   115→    async function load() {

   117→   117→        const res = await fetch(
   118→   118→          `/api/admin/subjects?classId=${filterClassId}`,
   119→   119→          { cache: "no-store" }

   122→   122→        const data = (await res.json()) as { subjects: AcademicSubject[] };
   123→   123→        if (cancelled) return;
   124→   124→        setSubjects(data.subjects);
   125→   125→        setFilterSubjectId("all");
   126→   126→      } catch {
   127→   127→        // ignore
   128→   128→      }
   129→   129→    }
   130→   130→    load();
   131→   131→    return () => {
   132→   132→      cancelled = true;
   133→   133→    };
   134→   134→  }, [filterClassId]);
   135→   135→
   136→   136→  // Load topics when filter subject changes (dependent dropdown)
   137→   137→  useEffect(() => {
   138→   138→    if (filterSubjectId === "all") {
   139→   139→      setTopics([]);
   140→   140→      setFilterTopicId("all");
   141→   141→      return;
   142→   142→    }
   143→   143→    let cancelled = false;
   144→   144→    async function load() {
   145→   145→      try {
   146→   146→        const res = await fetch(
   147→   147→          `/api/admin/topics?subjectId=${filterSubjectId}`,
   148→   148→          { cache: "no-store" }
   149→   149→        );

   152→   152→        if (cancelled) return;
   153→   153→        setTopics(data.topics);
   154→   154→        setFilterTopicId("all");
   155→   155→      } catch {
   156→   156→        // ignore
   157→   157→      }
   158→   158→    }
   159→   159→    load();
   160→   160→    return () => {
   161→   161→      cancelled = true;
   162→   162→    };
   163→   163→  }, [filterSubjectId]);
   164→   164→
   165→   165→  const load = useCallback(async () => {

   167→   167→    try {

   171→   171→        url.searchParams.set("subjectId", filterSubjectId);

   183→   183→      setLoading(false);
   184→   184→    }
   185→   185→  }, [filterClassId, filterSubjectId, filterTopicId, filterStatus, search]);
   186→   186→
   187→   187→  useEffect(() => {
   188→   188→    load();
   189→   189→  }, [load]);
   190→   190→
   191→   191→  async function togglePublish(q: QuestionWithChain) {
   192→   192→    const next: QuestionStatus = q.status === "published" ? "draft" : "published";
   193→   193→    try {
   194→   194→      const res = await fetch(`/api/admin/questions/${q.id}`, {
   195→   195→        method: "PATCH",
   196→   196→        headers: { "Content-Type": "application/json" },
   197→   197→        body: JSON.stringify({ status: next }),
   198→   198→      });
   199→   199→      if (!res.ok) {
   200→   200→        const d = await res.json().catch(() => ({}));
   201→   201→        throw new Error(d.error ?? "Failed to update status");
   202→   202→      }
   203→   203→      toast.success(
   204→   204→        next === "published" ? "Question published" : "Question moved to draft"
   205→   205→      );
   206→   206→      await load();
   207→   207→    } catch (e) {
   208→   208→      toast.error(e instanceof Error ? e.message : "Failed to update status");
   209→   209→    }
   210→   210→  }
   211→   211→
   212→   212→  const visible = (questions ?? []).slice(0, PAGE_SIZE);
   213→   213→  const hiddenCount = (questions?.length ?? 0) - visible.length;
   214→   214→
   215→   215→  return (
   216→   216→    <div className="space-y-6">
   217→   217→      <div className="flex flex-wrap items-center justify-between gap-3">
   218→   218→        <div>
   219→   219→          <h2 className="text-2xl font-bold text-slate-900">Questions</h2>
   220→   220→          <p className="text-sm text-slate-600 mt-1">
   221→   221→            Create, edit, and publish MCQs. Questions are grouped by topic.
   222→   222→          </p>
   223→   223→        </div>
   224→   224→        <div className="flex items-center gap-2">
   225→   225→          <Button variant="outline" size="sm" onClick={load} disabled={loading}>
   226→   226→            <RefreshCw className={loading ? "size-4 animate-spin" : "size-4"} />
   227→   227→            <span className="hidden sm:inline">Refresh</span>
   228→   228→          </Button>
   229→   229→          <Button
   230→   230→            size="sm"
   231→   231→            className="bg-emerald-500 hover:bg-emerald-600 text-white"
   232→   232→            onClick={() => setCreating(true)}
   233→   233→            disabled={classes.length === 0}
   234→   234→          >
   235→   235→            <Plus className="size-4" />
   236→   236→            Add Question
   237→   237→          </Button>
   238→   238→        </div>
   239→   239→      </div>
   240→   240→
   241→   241→      {classes.length === 0 && (
   242→   242→        <div className="rounded-md bg-amber-50 border border-amber-200 px-4 py-3 text-sm text-amber-800">
   243→   243→          You need to create at least one class, subject, and topic before you
   244→   244→          can add questions.
   245→   245→        </div>
   246→   246→      )}
   247→   247→
   248→   248→      {/* Filters */}
   249→   249→      <div className="flex flex-wrap gap-3">
   250→   250→        <div className="relative flex-1 min-w-[200px] max-w-sm">
   251→   251→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 size-4 text-slate-400 pointer-events-none" />
   252→   252→          <Input
   253→   253→            placeholder="Search question text…"
   254→   254→            value={search}
   255→   255→            onChange={(e) => setSearch(e.target.value)}
   256→   256→            className="pl-9"
   257→   257→          />
   258→   258→        </div>
   259→   259→        <Select value={filterClassId} onValueChange={setFilterClassId}>
   260→   260→          <SelectTrigger className="w-[160px]">
   261→   261→            <SelectValue placeholder="All classes" />
   262→   262→          </SelectTrigger>
   263→   263→          <SelectContent>
   264→   264→            <SelectItem value="all">All classes</SelectItem>
   265→   265→            {classes.map((c) => (
   266→   266→              <SelectItem key={c.id} value={c.id}>
   267→   267→                {c.name}
   268→   268→              </SelectItem>
   269→   269→            ))}
   270→   270→          </SelectContent>
   271→   271→        </Select>
   272→   272→        <Select
   273→   273→          value={filterSubjectId}
   274→   274→          onValueChange={setFilterSubjectId}
   275→   275→          disabled={filterClassId === "all"}
   276→   276→        >
   277→   277→          <SelectTrigger className="w-[160px]">
   278→   278→            <SelectValue placeholder="All subjects" />
   279→   279→          </SelectTrigger>
   280→   280→          <SelectContent>
   281→   281→            <SelectItem value="all">All subjects</SelectItem>
   282→   282→            {subjects.map((s) => (
   283→   283→              <SelectItem key={s.id} value={s.id}>
   284→   284→                {s.name}
   285→   285→              </SelectItem>
   286→   286→            ))}
   287→   287→          </SelectContent>
   288→   288→        </Select>
   289→   289→        <Select
   290→   290→          value={filterTopicId}
   291→   291→          onValueChange={setFilterTopicId}
   292→   292→          disabled={filterSubjectId === "all"}
   293→   293→        >
   294→   294→          <SelectTrigger className="w-[180px]">
   295→   295→            <SelectValue placeholder="All topics" />
   296→   296→          </SelectTrigger>
   297→   297→          <SelectContent>
   298→   298→            <SelectItem value="all">All topics</SelectItem>
   299→   299→            {topics.map((t) => (
   300→   300→              <SelectItem key={t.id} value={t.id}>
   301→   301→                {t.name}
   302→   302→              </SelectItem>
   303→   303→            ))}
   304→   304→          </SelectContent>
   305→   305→        </Select>
   306→   306→        <Select value={filterStatus} onValueChange={setFilterStatus}>
   307→   307→          <SelectTrigger className="w-[140px]">
   308→   308→            <SelectValue placeholder="All statuses" />
   309→   309→          </SelectTrigger>
   310→   310→          <SelectContent>
   311→   311→            <SelectItem value="all">All statuses</SelectItem>
   312→   312→            <SelectItem value="published">Published</SelectItem>
   313→   313→            <SelectItem value="draft">Draft</SelectItem>
   314→   314→          </SelectContent>
   315→   315→        </Select>
   316→   316→      </div>
   317→   317→
   318→   318→      {/* List */}
   319→   319→      {loading ? (
   320→   320→        <ListSkeleton rows={6} />
   321→   321→      ) : !questions || questions.length === 0 ? (
   322→   322→        <EmptyState
   323→   323→          icon={FileQuestion}
   324→   324→          title="No questions yet"
   325→   325→          description={
   326→   326→            classes.length === 0
   327→   327→              ? "Create a class, subject, and topic first, then add questions."
   328→   328→              : "Add your first question to start building the quiz bank."
   329→   329→          }
   330→   330→          action={
   331→   331→            classes.length > 0 ? (
   332→   332→              <Button
   333→   333→                size="sm"
   334→   334→                className="bg-emerald-500 hover:bg-emerald-600 text-white"
   335→   335→                onClick={() => setCreating(true)}
   336→   336→              >
   337→   337→                <Plus className="size-4" />
   338→   338→                Add Question
   339→   339→              </Button>
   340→   340→            ) : undefined
   341→   341→          }
   342→   342→        />
   343→   343→      ) : (
   344→   344→        <div className="rounded-xl border border-slate-200 bg-white overflow-hidden">
   345→   345→          <div className="overflow-x-auto">
   346→   346→            <table className="w-full text-sm">
   347→   347→              <thead className="bg-slate-50 text-xs uppercase tracking-wider text-slate-500">
   348→   348→                <tr>
   349→   349→                  <th className="px-4 py-3 text-left font-medium">Question</th>
   350→   350→                  <th className="px-4 py-3 text-left font-medium hidden lg:table-cell">
   351→   351→                    Class
   352→   352→                  </th>
   353→   353→                  <th className="px-4 py-3 text-left font-medium hidden md:table-cell">
   354→   354→                    Subject
   355→   355→                  </th>
   356→   356→                  <th className="px-4 py-3 text-left font-medium hidden sm:table-cell">
   357→   357→                    Topic
   358→   358→                  </th>
   359→   359→                  <th className="px-4 py-3 text-left font-medium">Status</th>
   360→   360→                  <th className="px-4 py-3 text-left font-medium hidden md:table-cell">
   361→   361→                    Created
   362→   362→                  </th>
   363→   363→                  <th className="px-4 py-3 text-right font-medium">Actions</th>
   364→   364→                </tr>
   365→   365→              </thead>
   366→   366→              <tbody className="divide-y divide-slate-100">
   367→   367→                {visible.map((q, idx) => (
   368→   368→                  <motion.tr
   369→   369→                    key={q.id}
   370→   370→                    initial={{ opacity: 0, y: 4 }}
   371→   371→                    animate={{ opacity: 1, y: 0 }}
   372→   372→                    transition={{ duration: 0.15, delay: Math.min(idx * 0.02, 0.3) }}
   373→   373→                    className="hover:bg-slate-50/60 align-top"
   374→   374→                  >
   375→   375→                    <td className="px-4 py-3 font-medium text-slate-900 max-w-md">
   376→   376→                      <div className="line-clamp-2">{q.questionText}</div>
   377→   377→                      <div className="mt-1 sm:hidden text-xs text-slate-500">
   378→   378→                        {q.topic.subject.class.name} → {q.topic.subject.name} →{" "}
   379→   379→                        {q.topic.name}
   380→   380→                      </div>
   381→   381→                    </td>
   382→   382→                    <td className="px-4 py-3 text-slate-600 hidden lg:table-cell">
   383→   383→                      {q.topic.subject.class.name}
   384→   384→                    </td>
   385→   385→                    <td className="px-4 py-3 text-slate-600 hidden md:table-cell">
   386→   386→                      {q.topic.subject.name}
   387→   387→                    </td>
   388→   388→                    <td className="px-4 py-3 text-slate-600 hidden sm:table-cell">
   389→   389→                      {q.topic.name}
   390→   390→                    </td>
   391→   391→                    <td className="px-4 py-3">
   392→   392→                      <StatusBadge
   393→   393→                        tone={q.status === "published" ? "published" : "draft"}
   394→   394→                        label={q.status === "published" ? "Published" : "Draft"}
   395→   395→                      />
   396→   396→                    </td>
   397→   397→                    <td className="px-4 py-3 text-slate-500 hidden md:table-cell text-xs">
   398→   398→                      {new Date(q.createdAt).toLocaleDateString(undefined, {
   399→   399→                        year: "numeric",
   400→   400→                        month: "short",
   401→   401→                        day: "numeric",
   402→   402→                      })}
   403→   403→                    </td>
   404→   404→                    <td className="px-4 py-3 text-right">
   405→   405→                      <div className="inline-flex items-center gap-1">
   406→   406→                        <Button
   407→   407→                          variant="ghost"
   408→   408→                          size="sm"
   409→   409→                          onClick={() => togglePublish(q)}
   410→   410→                          title={
   411→   411→                            q.status === "published" ? "Unpublish" : "Publish"
   412→   412→                          }
   413→   413→                          className={
   414→   414→                            q.status === "published"
   415→   415→                              ? "text-amber-600 hover:text-amber-700 hover:bg-amber-50"
   416→   416→                              : "text-emerald-600 hover:text-emerald-700 hover:bg-emerald-50"
   417→   417→                          }
   418→   418→                        >
   419→   419→                          {q.status === "published" ? (
   420→   420→                            <EyeOff className="size-4" />
   421→   421→                          ) : (
   422→   422→                            <Eye className="size-4" />
   423→   423→                          )}
   424→   424→                          <span className="hidden sm:inline">
   425→   425→                            {q.status === "published" ? "Unpublish" : "Publish"}
   426→   426→                          </span>
   427→   427→                        </Button>
   428→   428→                        <Button
   429→   429→                          variant="ghost"
   430→   430→                          size="sm"
   431→   431→                          onClick={() => setEditing(q)}
   432→   432→                        >
   433→   433→                          <Pencil className="size-4" />
   434→   434→                          <span className="hidden sm:inline">Edit</span>
   435→   435→                        </Button>
   436→   436→                        <ConfirmDialog
   437→   437→                          trigger={
   438→   438→                            <Button
   439→   439→                              variant="ghost"
   440→   440→                              size="sm"
   441→   441→                              className="text-red-600 hover:text-red-700 hover:bg-red-50"
   442→   442→                            >
   443→   443→                              <Trash2 className="size-4" />
   444→   444→                              <span className="hidden sm:inline">Delete</span>
   445→   445→                            </Button>
   446→   446→                          }
   447→   447→                          title="Delete this question?"
   448→   448→                          entityName={q.questionText.slice(0, 80) + (q.questionText.length > 80 ? "…" : "")}
   449→   449→                          onConfirm={async () => {
   450→   450→                            const res = await fetch(
   451→   451→                              `/api/admin/questions/${q.id}`,
   452→   452→                              { method: "DELETE" }
   453→   453→                            );
   454→   454→                            if (!res.ok) {
   455→   455→                              const d = await res.json().catch(() => ({}));
   456→   456→                              throw new Error(
   457→   457→                                d.error ?? "Failed to delete question"
   458→   458→                              );
   459→   459→                            }
   460→   460→                            toast.success("Question deleted");
   461→   461→                            await load();
   462→   462→                          }}
   463→   463→                        />
   464→   464→                      </div>
   465→   465→                    </td>
   466→   466→                  </motion.tr>
   467→   467→                ))}
   468→   468→              </tbody>
   469→   469→            </table>
   470→   470→          </div>
   471→   471→          {hiddenCount > 0 && (
   472→   472→            <div className="px-4 py-3 border-t border-slate-200 bg-slate-50 text-xs text-slate-500 text-center">
   473→   473→              Showing {visible.length} of {questions.length} questions. Narrow
   474→   474→              your filters to see more.
   475→   475→            </div>
   476→   476→          )}
   477→   477→        </div>
   478→   478→      )}
   479→   479→
   480→   480→      {/* Create / Edit dialog */}
   481→   481→      <QuestionDialog
   482→   482→        open={creating || editing !== null}
   483→   483→        editing={editing}
   484→   484→        classes={classes}
   485→   485→        onClose={() => {

   941→   941→function OptionField({
